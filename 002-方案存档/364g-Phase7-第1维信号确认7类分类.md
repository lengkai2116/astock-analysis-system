**当前三态**：
- `否决` → red
- `强确认` → green
- `基础确认` → green
- `未确认` → yellow（默认）

**问题**：
- 仅三态，无法区分"右侧确认"vs"趋势运行中"vs"左侧试探"等不同性质
- 无共振评分（多维度交叉确认的量化强度）
- 无衰减检测（信号随时间推移的失效程度）
- 无信号属性的7类全量分类

### 1.2 信号属性定义（359号 §1.3，已确认规格）

| 属性 | 代码 | 定义 | 判断优先级 |
|------|------|------|:---------:|
| **风险警示** | risk_warning | 风险因素触发，需警惕 | 1（最高） |
| **右侧确认** | right_confirmed | 趋势已被确认，多信号共振 | 2 |
| **右侧初现** | right_emerging | 趋势信号刚出现，尚未验证 | 3 |
| **左侧试探** | left_probing | 下跌中出现反转迹象，未确认 | 4 |
| **趋势运行中** | trend_running | 已有趋势在运行，无新信号 | 5 |
| **盘整待变** | consolidating | 无明确方向，等待突破 | 6 |
| **中性观望** | neutral | 无明显特征 | 7（默认） |

### 1.3 共振评分公式（359号 §1.4，已确认）

```
strength_score = structure_score × 0.25    # 结构确认度（缠论）
              + volume_score × 0.30        # 量价配合度（量价引擎）
              + chip_score × 0.25          # 筹码支持度（筹码引擎）
              + factor_score × 0.20        # 因子支撑度（因子引擎）
```

```python
"""364g Phase7：信号属性分类器（7类全量分类）

基于 359号 §1.3 规格实现，按优先级从高到低判定：
  1. risk_warning → 风险警示
  2. right_confirmed → 右侧确认
  3. right_emerging → 右侧初现
  4. left_probing → 左侧试探
  5. trend_running → 趋势运行中
  6. consolidating → 盘整待变
  7. neutral → 中性观望（默认）
"""
from __future__ import annotations
from dataclasses import dataclass, field
from typing import Dict, List, Optional
import logging

logger = logging.getLogger(__name__)

# ── 7类属性常量 ──
ATTR_RISK_WARNING = 'risk_warning'
ATTR_RIGHT_CONFIRMED = 'right_confirmed'
ATTR_RIGHT_EMERGING = 'right_emerging'
ATTR_LEFT_PROBING = 'left_probing'
ATTR_TREND_RUNNING = 'trend_running'
ATTR_CONSOLIDATING = 'consolidating'
ATTR_NEUTRAL = 'neutral'

ATTR_NAMES = {
    ATTR_RISK_WARNING: '风险警示',
    ATTR_RIGHT_CONFIRMED: '右侧确认',
    ATTR_RIGHT_EMERGING: '右侧初现',
    ATTR_LEFT_PROBING: '左侧试探',
    ATTR_TREND_RUNNING: '趋势运行中',
    ATTR_CONSOLIDATING: '盘整待变',
    ATTR_NEUTRAL: '中性观望',
}

# ── 属性灯映射 ──
ATTR_LIGHTS = {
    ATTR_RISK_WARNING: 'red',
    ATTR_RIGHT_CONFIRMED: 'green',
    ATTR_RIGHT_EMERGING: 'green',
    ATTR_LEFT_PROBING: 'yellow',
    ATTR_TREND_RUNNING: 'green',
    ATTR_CONSOLIDATING: 'yellow',
    ATTR_NEUTRAL: 'yellow',
}

# 判断优先级（从高到低）
ATTR_PRIORITY = [
    ATTR_RISK_WARNING, ATTR_RIGHT_CONFIRMED, ATTR_RIGHT_EMERGING,
    ATTR_LEFT_PROBING, ATTR_TREND_RUNNING, ATTR_CONSOLIDATING, ATTR_NEUTRAL,
]


@dataclass
class SignalAttribute:
    """信号属性分类结果"""
    code: str = ATTR_NEUTRAL
    name: str = '中性观望'
    confidence: float = 0.5
    evidence: List[str] = field(default_factory=list)
    light: str = 'yellow'

    def to_dict(self) -> dict:
        return {
            'code': self.code,
            'name': self.name,
            'confidence': round(self.confidence, 2),
            'evidence': self.evidence,
            'light': self.light,
        }


class SignalAttributeClassifier:
    """364g Phase7：7类信号属性分类器

    判定逻辑严格按 359号 §1.3 规格的优先级执行：
      风险优先 → 右侧优先 → 左侧次之 → 趋势再次 → 盘整兜底 → 中性默认
    """

    def classify(self, tags: dict, dims: dict,
                 lifecycle: dict = None,
                 strength_score: float = 0,
                 registered_hits: list = None) -> SignalAttribute:
        """主分类入口

        Args:
            tags: pre_feat_cache 扁平化标签
            dims: status_engine._build_dimensions 输出的维度 dict
            lifecycle: 信号生命周期 dict（_signal_lifecycle 输出）
            strength_score: 共振评分（0-100）
            registered_hits: signal_registry 检测到的触发信号列表

        Returns:
            SignalAttribute 分类结果
        """
        registered_hits = registered_hits or []
        try:
            return self._classify_impl(tags, dims, lifecycle, strength_score, registered_hits)
        except Exception as e:
            logger.debug("信号属性分类失败: %s", e)
            return SignalAttribute(code=ATTR_NEUTRAL, name='中性观望',
                                   confidence=0.3, evidence=[f'分类异常: {e}'])

    def _classify_impl(self, tags, dims, lifecycle, strength_score, registered_hits):
        # === 按优先级逐项判定 ===

        # 优先级1：风险警示
        attr = self._check_risk_warning(tags, dims)
        if attr:
            return attr

        # 优先级2：右侧确认
        attr = self._check_right_confirmed(tags, dims, lifecycle, strength_score, registered_hits)
        if attr:
            return attr

        # 优先级3：右侧初现
        attr = self._check_right_emerging(tags, dims, lifecycle, registered_hits)
        if attr:
            return attr

        # 优先级4：左侧试探
        attr = self._check_left_probing(tags, dims)
        if attr:
            return attr

        # 优先级5：趋势运行中
        attr = self._check_trend_running(tags, dims)
        if attr:
            return attr

        # 优先级6：盘整待变
        attr = self._check_consolidating(tags, dims)
        if attr:
            return attr

        # 优先级7：中性观望（默认）
        return SignalAttribute(
            code=ATTR_NEUTRAL, name='中性观望',
            confidence=0.5, evidence=['无明显特征'], light='yellow'
        )

    def _check_risk_warning(self, tags, dims) -> Optional[SignalAttribute]:
        """风险警示：风险三维评估等级≥高"""
        risk_state = (dims.get('risk') or {}).get('state', '')
        if risk_state == '高':
            return SignalAttribute(
                code=ATTR_RISK_WARNING, name='风险警示',
                confidence=0.7,
                evidence=[f'风险等级={risk_state}'],
                light='red'
            )
        return None

    def _check_right_confirmed(self, tags, dims, lifecycle, strength_score, hits) -> Optional[SignalAttribute]:
        """右侧确认：4个条件全部满足

        条件1：signal_registry匹配到≥1个右侧信号
        条件2：信号已通过验证期（3-5日不破信号价）
        条件3：共振维度≥2（结构+量价/筹码中至少2个score>0.5）
        条件4：风险等级<高
        """
        # 条件1：有右侧信号触发
        right_hits = [h for h in hits if h.get('type') in (
            'chan_third_buy', 'volume_breakout', 'ma_bullish', 'platform_breakout')]
        if not right_hits:
            return None

        # 条件2：信号已通过验证期
        verified = False
        if lifecycle:
            stage = lifecycle.get('stage', '')
            verified = stage in ('中期', '已延伸')  # 初期=刚触发未验证，中期=已验证

        # 条件3：共振维度≥2
        structure_state = (dims.get('structure') or {}).get('state', '')
        vp_state = (dims.get('vp') or {}).get('state', '')
        chip_state = (dims.get('chip_fund') or {}).get('state', '')
        resonance_count = sum(1 for s in [structure_state, vp_state, chip_state]
                              if s in ('上升', '健康', '强健康', '流入'))

        # 条件4：风险等级<高
        risk_low = (dims.get('risk') or {}).get('state', '') != '高'

        if right_hits and verified and resonance_count >= 2 and risk_low:
            conf = min(0.95, 0.6 + strength_score / 500 + resonance_count * 0.05)
            return SignalAttribute(
                code=ATTR_RIGHT_CONFIRMED, name='右侧确认',
                confidence=conf,
                evidence=[
                    f'信号触发: {right_hits[0].get("name", "")}',
                    f'已验证(lifecycle={lifecycle.get("stage", "") if lifecycle else "N/A"})',
                    f'共振维度={resonance_count}',
                    f'风险等级<高',
                ],
                light='green'
            )
        return None

    def _check_right_emerging(self, tags, dims, lifecycle, hits) -> Optional[SignalAttribute]:
        """右侧初现：信号刚出现，尚未验证

        条件1：signal_registry匹配到≥1个右侧信号
        条件2：信号未通过验证期（触发<3日或已破信号价）
        条件3：共振维度≥1
        """
        right_hits = [h for h in hits if h.get('type') in (
            'chan_third_buy', 'volume_breakout', 'ma_bullish', 'platform_breakout')]
        if not right_hits:
            return None

        # 条件2：未通过验证期
        not_verified = True
        if lifecycle:
            stage = lifecycle.get('stage', '')
            not_verified = stage in ('初期', '回撤', None, '')

        # 条件3：至少1维共振
        structure_up = (dims.get('structure') or {}).get('state', '') == '上升'
        vp_healthy = (dims.get('vp') or {}).get('state', '') in ('健康', '强健康')
        chip_in = (dims.get('chip_fund') or {}).get('state', '') == '流入'
        resonance_count = sum(1 for b in [structure_up, vp_healthy, chip_in] if b)

        if right_hits and not_verified and resonance_count >= 1:
            return SignalAttribute(
                code=ATTR_RIGHT_EMERGING, name='右侧初现',
                confidence=0.6,
                evidence=[
                    f'信号触发: {right_hits[0].get("name", "")}',
                    f'未通过验证期',
                    f'共振维度={resonance_count}',
                ],
                light='green'
            )
        return None

    def _check_left_probing(self, tags, dims) -> Optional[SignalAttribute]:
        """左侧试探：5个条件全部必须满足

        条件1：缠论趋势=下降
        条件2：筹码流入强度>0.5
        条件3：量价健康度>60
        条件4：估值=低估/极度低估
        条件5：风险等级<高
        """
        structure_down = (dims.get('structure') or {}).get('state', '') == '下降'
        chip_in = (dims.get('chip_fund') or {}).get('state', '') == '流入'
        vp_healthy = (dims.get('vp') or {}).get('state', '') in ('健康', '强健康')
        valuation = (dims.get('valuation') or {}).get('state', '')
        val_undervalued = valuation in ('低估', '极度低估')
        risk_low = (dims.get('risk') or {}).get('state', '') != '高'

        if structure_down and chip_in and vp_healthy and val_undervalued and risk_low:
            return SignalAttribute(
                code=ATTR_LEFT_PROBING, name='左侧试探',
                confidence=0.5,
                evidence=[
                    '缠论趋势下降+筹码流入+量价健康+估值低估+低风险',
                    '左侧反转迹象（5条件全满足）',
                ],
                light='yellow'
            )
        return None

    def _check_trend_running(self, tags, dims) -> Optional[SignalAttribute]:
        """趋势运行中：已有趋势在运行，无新信号

        条件1：结构强度>0.6（趋势明确）
        条件2：无新信号触发
        条件3：量价健康度>50
        条件4：风险等级<中
        """
        structure_up = (dims.get('structure') or {}).get('state', '') == '上升'
        vp_healthy = (dims.get('vp') or {}).get('state', '') in ('健康', '强健康', '中性')
        risk_not_high = (dims.get('risk') or {}).get('state', '') != '高'
        # 无活跃信号
        time_state = (dims.get('time') or {}).get('state', '')
        no_new_signal = time_state in ('中期', '已延伸')  # 有运行中但无新信号

        if structure_up and vp_healthy and risk_not_high and no_new_signal:
            return SignalAttribute(
                code=ATTR_TREND_RUNNING, name='趋势运行中',
                confidence=0.65,
                evidence=[
                    f'结构上升趋势明确',
                    f'量价健康',
                    f'信号持续运行中(lifecycle={time_state})',
                ],
                light='green'
            )
        return None

    def _check_consolidating(self, tags, dims) -> Optional[SignalAttribute]:
        """盘整待变：无明确方向，等待突破

        条件1：结构方向=盘整
        条件2：量价健康度在中性区间
        条件3：筹码无明显流入/流出
        条件4：无信号触发
        """
        structure_range = (dims.get('structure') or {}).get('state', '') == '盘整'
        vp_neutral = (dims.get('vp') or {}).get('state', '') == '中性'
        chip_neutral = (dims.get('chip_fund') or {}).get('state', '') == '中性'
        time_neutral = (dims.get('time') or {}).get('state', '') == '中期'

        if structure_range and vp_neutral and chip_neutral:
            return SignalAttribute(
                code=ATTR_CONSOLIDATING, name='盘整待变',
                confidence=0.5,
                evidence=['结构盘整+量价中性+筹码中性，等待方向选择'],
                light='yellow'
            )
        return None
```

### 2.2 新建文件：signal_decay_detector.py

**位置**：`backend/app/opportunity_atlas/signal_decay_detector.py`

```python
"""364g Phase7：信号衰减检测器（5维度联合衰减）

基于 359号 §1.5 规格实现，输出 0-100 衰减分。

衰减分映射：
  0-39  → healthy（信号健康）
  40-69 → fading（信号衰减）
  70-100 → broken（信号失效）
"""
from __future__ import annotations
from dataclasses import dataclass, field
from typing import Dict, List, Optional
import logging

logger = logging.getLogger(__name__)

# 衰减分状态常量
DECAY_HEALTHY = 'healthy'
DECAY_FADING = 'fading'
DECAY_BROKEN = 'broken'


@dataclass
class DecayBreakdown:
    """衰减评分明细"""
    price_trend: Dict = field(default_factory=lambda: {'status': 'stable', 'score': 0})
    volume_price: Dict = field(default_factory=lambda: {'status': 'healthy', 'score': 0})
    volume_energy: Dict = field(default_factory=lambda: {'status': 'normal', 'score': 0})
    chip_change: Dict = field(default_factory=lambda: {'status': 'accumulating', 'score': 0})
    main_force: Dict = field(default_factory=lambda: {'status': 'accumulating', 'score': 0})


@dataclass
class DecayResult:
    """衰减检测结果"""
    overall_score: int = 0
    overall_status: str = DECAY_HEALTHY
    breakdown: DecayBreakdown = field(default_factory=DecayBreakdown)
    evidence: List[str] = field(default_factory=list)
    lifecycle_day: int = 0
    lifecycle_stage: str = '初期'
    verified: bool = False
    distance_to_signal_pct: str = ''

    def to_dict(self) -> dict:
        return {
            'overall_score': self.overall_score,
            'overall_status': self.overall_status,
            'breakdown': {
                'price_trend': self.breakdown.price_trend,
                'volume_price': self.breakdown.volume_price,
                'volume_energy': self.breakdown.volume_energy,
                'chip_change': self.breakdown.chip_change,
                'main_force': self.breakdown.main_force,
            },
            'evidence': self.evidence,
            'lifecycle_day': self.lifecycle_day,
            'lifecycle_stage': self.lifecycle_stage,
            'verified': self.verified,
            'distance_to_signal_pct': self.distance_to_signal_pct,
        }


class SignalDecayDetector:
    """364g Phase7：信号衰减检测器

    5维度联合检测，权重分配：
      走势变化 30% + 量价关系 25% + 量能变化 20% + 筹码变化 15% + 主力行为 10%
    """

    def detect(self, tags: dict, lifecycle: dict = None,
               daily_data: dict = None,
               moneyflow_data: dict = None,
               lhb_data: dict = None,
               margin_data: dict = None) -> DecayResult:
        """检测信号衰减

        Args:
            tags: pre_feat_cache 扁平化标签
            lifecycle: 信号生命周期（_signal_lifecycle 输出）
            daily_data: 日线数据（近5日 OHLCV）{'closes':[], 'volumes':[], 'highs':[], 'lows':[]}
            moneyflow_data: 资金流向 {'net_lg_amount': float, 'net_sm_amount': float, ...}
            lhb_data: 龙虎榜 {'institutional_buy': bool, 'institutional_sell': bool}
            margin_data: 融资 {'margin_change_5d': float}

        Returns:
            DecayResult
        """
        result = DecayResult()
        if not lifecycle:
            result.evidence = ['无活跃信号，衰减检测跳过']
            return result

        # 生命周期基本信息
        result.lifecycle_stage = lifecycle.get('stage', '中期')
        dist_pct = lifecycle.get('dist_pct', 0)
        result.distance_to_signal_pct = f'{dist_pct:+.1f}%'
        result.verified = lifecycle.get('stage', '') in ('中期', '已延伸')
        # 估算信号天数（基于 lifecycle 阶段）
        if lifecycle.get('stage') == '初期':
            result.lifecycle_day = 2
        elif lifecycle.get('stage') == '中期':
            result.lifecycle_day = 7
        elif lifecycle.get('stage') == '已延伸':
            result.lifecycle_day = 15
        else:
            result.lifecycle_day = 0

        breakdown = DecayBreakdown()

        # ── 维度1：走势变化（30%）──
        # 检测：价格vs信号价距离变化（3日趋势）
        price_score = self._calc_price_trend_decay(dist_pct, daily_data)
        breakdown.price_trend = price_score

        # ── 维度2：量价关系（25%）──
        # 检测：量比变化+量价配合度
        vp_score = self._calc_vp_decay(tags, daily_data)
        breakdown.volume_price = vp_score

        # ── 维度3：量能变化（20%）──
        # 检测：成交量vs20日均量比值
        energy_score = self._calc_volume_energy_decay(daily_data)
        breakdown.volume_energy = energy_score

        # ── 维度4：筹码变化（15%）──
        # 检测：主力资金流向+筹码集中度变化
        chip_score = self._calc_chip_decay(tags, moneyflow_data)
        breakdown.chip_change = chip_score

        # ── 维度5：主力行为（10%）──
        # 检测：龙虎榜席位+融资余额变化
        force_score = self._calc_main_force_decay(lhb_data, margin_data)
        breakdown.main_force = force_score

        # ── 加权合成 ──
        overall = int(
            price_score['score'] * 0.30 +
            vp_score['score'] * 0.25 +
            energy_score['score'] * 0.20 +
            chip_score['score'] * 0.15 +
            force_score['score'] * 0.10
        )
        result.overall_score = min(100, max(0, overall))

        # 状态映射
        if result.overall_score <= 39:
            result.overall_status = DECAY_HEALTHY
        elif result.overall_score <= 69:
            result.overall_status = DECAY_FADING
        else:
            result.overall_status = DECAY_BROKEN

        # 证据链
        result.breakdown = breakdown
        result.evidence = self._build_evidence(result)
        return result

    def _calc_price_trend_decay(self, dist_pct, daily_data) -> dict:
        """走势变化衰减（30%权重）

        衰减信号：价格vs信号价距离连续3日缩小（利润回吐）
        """
        score = 0
        status = 'stable'
        evidence = ''

        if dist_pct < 0:
            # 已跌破信号价
            score = 80
            status = 'broken'
            evidence = f'已跌破信号价({dist_pct:+.1f}%)'
        elif dist_pct < 3:
            # 距信号价很近，利润薄
            score = 40
            status = 'fading'
            evidence = f'利润微薄({dist_pct:+.1f}%)'
        else:
            # 检查3日趋势
            if daily_data and daily_data.get('closes'):
                closes = daily_data['closes']
                if len(closes) >= 4:
                    # 3日价格趋势
                    chg_3d = (closes[-1] / closes[-4] - 1) * 100 if closes[-4] > 0 else 0
                    if chg_3d < -3:
                        score = 30
                        status = 'fading'
                        evidence = f'3日下跌{chg_3d:.1f}%，利润回吐'
                    elif chg_3d < -5:
                        score = 60
                        status = 'fading'
                        evidence = f'3日下跌{chg_3d:.1f}%，加速回吐'
            if not evidence:
                score = 0
                status = 'stable'
                evidence = f'价格趋势稳定({dist_pct:+.1f}%)'

        return {'status': status, 'score': score, 'evidence': evidence}

    def _calc_vp_decay(self, tags, daily_data) -> dict:
        """量价关系衰减（25%权重）

        衰减信号：放量后持续缩量+背离
        """
        score = 0
        status = 'healthy'
        evidence = ''

        # 从标签读取量价状态
        vp_fit = str(tags.get('volume_price_fit', ''))
        if vp_fit in ('diverging', 'severe_diverging'):
            score = 70
            status = 'diverging'
            evidence = f'量价背离({vp_fit})'
        elif vp_fit in ('healthy', 'strong_healthy'):
            score = 0
            status = 'healthy'
            evidence = '量价健康'
        else:
            # 从日线数据检测缩量趋势
            if daily_data and daily_data.get('volumes'):
                vols = daily_data['volumes']
                if len(vols) >= 5:
                    vol_ma5 = sum(vols[-5:]) / 5
                    vol_ma20 = sum(vols[-20:]) / 20 if len(vols) >= 20 else vol_ma5
                    if vol_ma5 < vol_ma20 * 0.6:
                        score = 40
                        status = 'shrinking'
                        evidence = '持续缩量（5日均量<20日均量60%）'
                    else:
                        score = 10
                        status = 'normal'
                        evidence = '量价关系正常'
            if not evidence:
                score = 10
                status = 'normal'
                evidence = '量价关系中性'

        return {'status': status, 'score': score, 'evidence': evidence}

    def _calc_volume_energy_decay(self, daily_data) -> dict:
        """量能变化衰减（20%权重）

        衰减信号：连续3日低于均量
        """
        score = 0
        status = 'normal'
        evidence = ''

        if daily_data and daily_data.get('volumes'):
            vols = daily_data['volumes']
            if len(vols) >= 23:
                vol_ma20 = sum(vols[-21:-1]) / 20
                # 最近3日成交量趋势
                recent_3d = vols[-3:]
                below_count = sum(1 for v in recent_3d if v < vol_ma20 * 0.7)
                if below_count >= 3:
                    score = 60
                    status = 'depleted'
                    evidence = '连续3日低于均量70%'
                elif below_count >= 2:
                    score = 30
                    status = 'weakening'
                    evidence = '2日低于均量70%'
                else:
                    score = 0
                    status = 'normal'
                    evidence = '量能正常'
        if not evidence:
            score = 10
            status = 'normal'
            evidence = '量能数据不足'

        return {'status': status, 'score': score, 'evidence': evidence}

    def _calc_chip_decay(self, tags, moneyflow_data) -> dict:
        """筹码变化衰减（15%权重）

        衰减信号：主力净流出+筹码分散
        """
        score = 0
        status = 'accumulating'
        evidence = ''

        if moneyflow_data:
            net_lg = moneyflow_data.get('net_lg_amount', 0)
            if net_lg and net_lg < 0:
                score = 50
                status = 'outflow'
                evidence = f'主力5日净流出{abs(net_lg):.2f}亿'
            else:
                score = 0
                status = 'accumulating'
                evidence = '主力资金中性/流入'

        # 筹码集中度变化
        chip_change = tags.get('chip_concentration_change', 0)
        try:
            chip_change = float(chip_change)
        except (TypeError, ValueError):
            chip_change = 0
        if chip_change < -0.1:
            score = max(score, 60)
            status = 'dispersing'
            evidence = f'筹码分散(变化{chip_change:.2f})'

        if not evidence:
            score = 10
            status = 'neutral'
            evidence = '筹码数据不足'

        return {'status': status, 'score': score, 'evidence': evidence}

    def _calc_main_force_decay(self, lhb_data, margin_data) -> dict:
        """主力行为衰减（10%权重）

        衰减信号：机构卖出+融资下降
        """
        score = 0
        status = 'neutral'
        evidence = ''

        if lhb_data:
            if lhb_data.get('institutional_sell'):
                score = 60
                status = 'selling'
                evidence = '龙虎榜机构卖出'
            elif lhb_data.get('institutional_buy'):
                score = 0
                status = 'buying'
                evidence = '龙虎榜机构买入'

        if margin_data:
            margin_chg = margin_data.get('margin_change_5d', 0)
            try:
                margin_chg = float(margin_chg)
            except (TypeError, ValueError):
                margin_chg = 0
            if margin_chg < -5:
                score = max(score, 40)
                status = 'leverage_declining'
                evidence = f'融资余额5日下降{margin_chg:.1f}%'

        if not evidence:
            score = 5
            status = 'neutral'
            evidence = '主力数据不足'

        return {'status': status, 'score': score, 'evidence': evidence}

    def _build_evidence(self, result: DecayResult) -> list:
        """构建完整证据链"""
        ev = []
        bd = result.breakdown
        for dim_name, dim_data in [
            ('走势', bd.price_trend), ('量价', bd.volume_price),
            ('量能', bd.volume_energy), ('筹码', bd.chip_change),
            ('主力', bd.main_force)
        ]:
            e = dim_data.get('evidence', '')
            if e:
                ev.append(f'【{dim_name}】{e}')
        ev.append(f'衰减总分: {result.overall_score}/100 ({result.overall_status})')
        return ev
```
