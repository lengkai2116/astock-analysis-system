### 1.2 confidence 评分实现（volume_price_strategy.py:3840-3870）

```python
def calc_confidence(self, fast_state: str, slow_dist: Dict[str, float],
                    matched_patterns: List[str]) -> float:
    """置信度由三个因素决定：
    1. 慢线频率：该状态在过去 5 天的占比 (权重 40%)
    2. 子形态命中率：匹配到的形态数 / 该状态注册的形态数 (权重 35%)
    3. 方向一致性：命中的形态是否与状态方向一致 (权重 25%)
    """
    slow_base = slow_dist.get(fast_state, 0.2)
    registered = STATE_PATTERN_REGISTRY.get(fast_state, [])
    if registered and matched_patterns:
        hits = sum(1 for p in matched_patterns if any(rp in p for rp in registered))
        hit_rate = hits / len(registered)
    else:
        hit_rate = 0.0
    state_level = STATE_LEVEL_MAP.get(fast_state, "NEUTRAL")
    consistent_hits = sum(1 for p in matched_patterns
                          if ("预涨" in p and state_level == "BULLISH")
                          or ("预跌" in p and state_level == "BEARISH"))
    total_dir_hits = sum(1 for p in matched_patterns if "预涨" in p or "预跌" in p)
    consistency = consistent_hits / max(total_dir_hits, 1)
    confidence = slow_base * 0.4 + hit_rate * 0.35 + consistency * 0.25
    return round(min(1.0, max(0.1, confidence)), 4)
```

**问题**：
- 输出为 0-1 浮点数，非标准化 10 分制
- 缺少 RPS（相对价格强度）因子
- 缺少筹码集中度因子
- 无量能强度（vol_ratio 分级）和背离惩罚因子

```python
def vp_score_10(self, fast_state: str, slow_dist: Dict[str, float],
                matched_patterns: List[str], vol_ratio: float = 1.0,
                has_divergence: bool = False, ma_alignment: str = 'mixed',
                chip_concentration: float = None,
                indicator_verify: Dict = None) -> Dict:
    """364e Phase5：量价健康度 10 分制评分

    评分公式（-2 ~ +10 基础分，映射为 0-100 百分制）：
      base = vp_pattern_score       # 量价形态得分（-2 ~ +3）
            + volume_energy_score   # 量能强度得分（0 ~ +2）
            + ma_alignment_score    # 均线排列得分（0 ~ +1）
            + chip_density_score    # 筹码密集得分（0 ~ +1）
            + indicator_score       # 指标验证得分（0 ~ +1）
            + divergence_penalty    # 背离惩罚（-2 ~ 0）

    Args:
        fast_state: VPStateMachine 快线判定结果
        slow_dist: 慢线 5 日分布
        matched_patterns: EnhancedPatternDetector 匹配的形态列表
        vol_ratio: 量比（最新成交量 / 20日均量）
        has_divergence: 是否存在量价背离
        ma_alignment: 均线排列（'bullish'/'bearish'/'mixed'）
        chip_concentration: 筹码集中度（0-1），None 表示无数据
        indicator_verify: 指标验证结果 {'macd_confirm':bool, 'boll_position':str, 'kdj_status':str}

    Returns:
        {
            'score': int,           # 10分制得分（0-10）
            'score_100': int,       # 百分制得分（0-100），= score * 10
            'level': str,           # 'strong'/'healthy'/'neutral'/'weak'/'dangerous'
            'breakdown': {          # 评分明细
                'vp_pattern': int,  # 量价形态分（-2 ~ +3）
                'volume_energy': int,# 量能强度分（0 ~ +2）
                'ma_alignment': int,# 均线排列分（0 ~ +1）
                'chip_density': int,# 筹码密集分（0 ~ +1）
                'indicator': int,   # 指标验证分（0 ~ +1）
                'divergence_penalty': int  # 背离惩罚（-2 ~ 0）
            },
            'evidence': list[str],  # 评分证据链
        }
    """
    breakdown = {}
    evidence = []

    # ── 1. 量价形态得分（-2 ~ +3）──
    # 黑马型/预涨型形态 → +3
    # 价涨量增(VP-1) / 放量突破(VP_BREAKOUT) → +1
    # 健康回调(VP-9 in UPTREND) → +1
    # 预跌型形态 → -2
    # 价跌量增(VP-7) / 价跌量缩(VP-9 in downtrend) → -1
    vp_pattern_score = 0
    bullish_patterns = [p for p in matched_patterns if '预涨' in p or '黑马' in p]
    bearish_patterns = [p for p in matched_patterns if '预跌' in p]
    STATE_BONUS = {
        VP_HEALTHY_BULL: 1, VP_BREAKOUT: 1, VP_PULLBACK: 1,
        VP_EXTREME_BULL: 1, VP_DIVERGE_BEAR: 0,
        VP_HEALTHY_BEAR: -1, VP_EXTREME_BEAR: -1, VP_DIVERGE_BULL: -1,
    }
    vp_pattern_score = STATE_BONUS.get(fast_state, 0)
    if bullish_patterns:
        vp_pattern_score += min(len(bullish_patterns), 2)  # 最多+2（与状态分合计最多+3）
        evidence.append(f"量价形态预涨{len(bullish_patterns)}种")
    if bearish_patterns:
        vp_pattern_score -= min(len(bearish_patterns), 2)
        evidence.append(f"量价形态预跌{len(bearish_patterns)}种")
    vp_pattern_score = max(-2, min(3, vp_pattern_score))
    breakdown['vp_pattern'] = vp_pattern_score

    # ── 2. 量能强度得分（0 ~ +2）──
    # 温和放量(1.2~3.0) → +1；显著放量(3.0~5.0) → +2
    # 萎缩(0.5~0.8) → +0；极度萎缩(<0.5) → +0（但不扣分）
    # 极度放量(>5) → +1（极端信号，非普通放量）
    volume_energy_score = 0
    if 1.2 <= vol_ratio <= 3.0:
        volume_energy_score = 1
        evidence.append(f"温和放量(量比{vol_ratio:.1f})")
    elif 3.0 < vol_ratio <= 5.0:
        volume_energy_score = 2
        evidence.append(f"显著放量(量比{vol_ratio:.1f})")
    elif vol_ratio > 5.0:
        volume_energy_score = 1
        evidence.append(f"极度放量(量比{vol_ratio:.1f})，注意极端风险")
    breakdown['volume_energy'] = volume_energy_score

    # ── 3. 均线排列得分（0 ~ +1）──
    # 多头排列 → +1
    # 空头/混合 → +0
    ma_alignment_score = 1 if ma_alignment == 'bullish' else 0
    if ma_alignment_score:
        evidence.append("均线多头排列")
    breakdown['ma_alignment'] = ma_alignment_score

    # ── 4. 筹码密集得分（0 ~ +1）──
    # 集中度 > 0.5 → +1（筹码单峰密集）
    chip_density_score = 0
    if chip_concentration is not None and chip_concentration > 0.5:
        chip_density_score = 1
        evidence.append(f"筹码密集(集中度{chip_concentration:.2f})")
    breakdown['chip_density'] = chip_density_score

    # ── 5. 指标验证得分（0 ~ +1）──
    # MACD金叉确认 或 BOLL中轨上方 或 KDJ金叉 → +1
    indicator_score = 0
    if indicator_verify:
        macd_ok = indicator_verify.get('macd_confirm', False)
        boll_ok = indicator_verify.get('boll_position', '') in ('中轨上方', '上轨')
        kdj_ok = indicator_verify.get('kdj_status', '') in ('金叉', '多头')
        if macd_ok or boll_ok or kdj_ok:
            indicator_score = 1
            confirms = []
            if macd_ok: confirms.append('MACD')
            if boll_ok: confirms.append('BOLL')
            if kdj_ok: triggers.append('KDJ')
            evidence.append(f"指标确认({'+'.join(confirms)})")
    breakdown['indicator'] = indicator_score

    # ── 6. 背离惩罚（-2 ~ 0）──
    # 顶背离 → -2（严重衰减信号）
    # 底背离 → 0（反转信号，不惩罚）
    divergence_penalty = -2 if has_divergence else 0
    if has_divergence:
        evidence.append("背离惩罚(-2)")
    breakdown['divergence_penalty'] = divergence_penalty

    # ── 合成 10 分制 ──
    raw_score = sum(breakdown.values())  # -2 ~ +8
    # 映射：raw_score 的理论范围是 -2 ~ +8，映射到 0 ~ 10
    # raw=-2 → 0分, raw=0 → 3分, raw=3 → 6分, raw=5 → 8分, raw=8 → 10分
    score_10 = max(0, min(10, round(raw_score * 10 / 8 + 2.5)))

    # 等级映射
    if score_10 >= 8:
        level = 'strong'
    elif score_10 >= 6:
        level = 'healthy'
    elif score_10 >= 4:
        level = 'neutral'
    elif score_10 >= 2:
        level = 'weak'
    else:
        level = 'dangerous'

    return {
        'score': score_10,
        'score_100': score_10 * 10,
        'level': level,
        'breakdown': breakdown,
        'evidence': evidence,
    }
```

```python
    - 快线层：单日量价状态 → 每日更新 → 划分8种状态（已有）

        # ... (现有实现不变)

    def calc_slow(self, n_days=5) -> Dict[str, float]:
        """慢线：过去N天快线结果的频率分布 — 保持不变"""
        # ... (现有实现不变)

    def classify_background(self, index_closes: np.ndarray,
                           index_volumes: np.ndarray) -> str:
        """364e 新增：背景层——市场整体量价状态

        基于指数（如沪深300/创业板指）的量价关系，判定市场背景：
        - favorable：市场整体量价健康 → 个股看涨信号加权
        - unfavorable：市场整体量价恶化 → 个股看涨信号降权
        - neutral：市场无明确方向 → 不加权不降权

        Args:
            index_closes: 指数收盘价序列
            index_volumes: 指数成交量序列

        Returns:
            'favorable' / 'unfavorable' / 'neutral'
        """
        if len(index_closes) < 20 or len(index_volumes) < 20:
            return 'neutral'

        # 指数 5日涨幅
        idx_chg_5d = (index_closes[-1] / index_closes[-5] - 1) * 100 if len(index_closes) >= 5 else 0
        # 指数量比（最新/20日均）
        idx_vol_ma20 = float(np.mean(index_volumes[-20:]))
        idx_vol_ratio = index_volumes[-1] / max(idx_vol_ma20, 1e-9)
        # 指数站上20日均线
        idx_ma20 = float(np.mean(index_closes[-20:]))
        idx_above_ma20 = index_closes[-1] > idx_ma20

        # favorable：指数放量上涨或缩量不跌
        if idx_chg_5d > 0 and idx_vol_ratio > 1.0 and idx_above_ma20:
            return 'favorable'
        if idx_chg_5d > -1 and idx_chg_5d < 1 and idx_vol_ratio < 0.8:
            return 'favorable'  # 缩量横盘=稳定

        # unfavorable：指数放量下跌
        if idx_chg_5d < -2 and idx_vol_ratio > 1.5:
            return 'unfavorable'
        if not idx_above_ma20 and idx_chg_5d < -1:
            return 'unfavorable'

        return 'neutral'

    def calc_confidence(self, fast_state, slow_dist, matched_patterns) -> float:
        """置信度（保持不变）"""
        # ... (现有实现不变)

    def vp_score_10(self, fast_state, slow_dist, matched_patterns,
                    vol_ratio=1.0, has_divergence=False, ma_alignment='mixed',
                    chip_concentration=None, indicator_verify=None) -> Dict:
        """364e 新增：10分制评分（见 §2.1 完整实现）"""
        # ... (完整实现见 §2.1)

    def define(self, closes, volumes, highs=None, lows=None,
               matched_patterns=None, n_days=5,
               index_closes=None, index_volumes=None) -> Dict:
        """执行完整状态判定（快线 + 慢线 + 背景层 + 置信度）

        364e Phase5 扩展：新增 background 状态输出。
        """
        matched_patterns = matched_patterns or []

        # 快线
        fast_state = self.classify_fast(closes, volumes, highs, lows)
        self.update_fast(fast_state)

        # 慢线
        slow_dist = self.calc_slow(n_days)

        # 背景层（新增）
        bg_state = self.classify_background(
            index_closes, index_volumes) if index_closes is not None else 'neutral'
        self._bg_state = bg_state

        # 背景过滤：市场不利时，看涨快线降权
        effective_state = fast_state
        if bg_state == 'unfavorable':
            # 仅保留看涨状态→降级为中性；看跌/中性不变
            if STATE_LEVEL_MAP.get(fast_state) == 'BULLISH':
                effective_state = VP_HEALTHY_BULL  # 保持状态但后续置信度降权

        # 置信度
        confidence = self.calc_confidence(fast_state, slow_dist, matched_patterns)
        # 背景降权
        if bg_state == 'unfavorable':
            confidence *= 0.8
        elif bg_state == 'favorable':
            confidence *= 1.1
        confidence = round(min(1.0, max(0.1, confidence)), 4)

        # 状态转换
        transition = self.check_transition(closes, volumes, fast_state)

        # 形态匹配
        if matched_patterns:
            registered = STATE_PATTERN_REGISTRY.get(fast_state, [])
            pattern_hits = [p for p in matched_patterns if any(rp in p for rp in registered)]
            pattern_hits.extend(p for p in matched_patterns if p not in pattern_hits
                                and ("预涨" in p or "预跌" in p))
        else:
            pattern_hits = matched_patterns or []

        return {
            'state': fast_state,
            'state_label': fast_state,
            'level': STATE_LEVEL_MAP.get(fast_state, "NEUTRAL"),
            'confidence': confidence,
            'slow_dist': slow_dist,
            'fast_history': self._fast_history[-n_days:] if self._fast_history else [],
            'pattern_hits': pattern_hits[:5],
            'transition': transition,
            'background': bg_state,  # 364e 新增
        }
```

```python
@dataclass
class StatusRecognition:
    """现状识别 — 描述当前市场状态，非买卖建议"""
    state: str = "UNKNOWN"
    state_label: str = ""
    trend: Dict = field(default_factory=lambda: {"direction": "", "strength": "", "stage": ""})
    momentum: Dict = field(default_factory=lambda: {"level": "", "score": 0.0})
    volume: Dict = field(default_factory=lambda: {"state": "", "structure": ""})
    support_resistance: Dict = field(default_factory=lambda: {"support": 0.0, "resistance": 0.0})
    risk_level: str = "MEDIUM"
    # 364e 新增：10分制量价健康度
    vp_health_score: Dict = field(default_factory=dict)
    # 364e 新增：三层状态机背景
    vp_background: str = "neutral"

    def to_dict(self) -> Dict:
        d = {
            'state': self.state,
            'state_label': self.state_label,
            'trend': self.trend,
            'momentum': self.momentum,
            'volume': self.volume,
            'support_resistance': self.support_resistance,
            'risk_level': self.risk_level,
        }
        if self.vp_health_score:
            d['vp_health_score'] = self.vp_health_score
        if self.vp_background:
            d['vp_background'] = self.vp_background
        return d
```
