```python
def calculate_single_factor(self, data: pd.DataFrame,
                            factor_name: str,
                            ts_code: str = None,
                            **kwargs) -> Optional[pd.Series]:
    """
    计算单个因子
    优先读取 factor_cache（需提供 ts_code），无缓存时实时计算
    """
    # 缓存优先：如果提供了 ts_code，尝试从 factor_cache 读取
    if ts_code is not None:
        try:
            from app.data import get_data_manager
            dm = get_data_manager()
            cached = dm.get_cached_factors(ts_code, [factor_name])
            if cached is not None and not cached.empty:
                # 缓存有该因子数据，按 trade_date 对齐返回
                cached = cached[cached['factor_name'] == factor_name]
                if not cached.empty:
                    # 将缓存数据映射回 Series（按 trade_date 排序）
                    cached = cached.sort_values('trade_date')
                    result = cached.set_index('trade_date')['value']
                    logger.debug(f"factor_cache 命中: {factor_name}")
                    return result
        except Exception as e:
            logger.debug(f"factor_cache 读取失败({factor_name}): {e}")

    # 缓存未命中 → 实时计算
    factor = self.registry.get_factor(factor_name, **kwargs)
    if factor is None:
        logger.error(f"未找到因子: {factor_name}")
        return None

    if not factor.check_data(data):
        logger.error(f"数据不满足因子 {factor_name} 的要求")
        return None

    try:
        result = factor.calculate(data)
        return result
    except Exception as e:
        logger.error(f"计算因子 {factor_name} 失败: {e}")
        return None
```

### Task 2: P0-1 — Vibe S1 市场情绪周期策略

**Covers:** 待解决事项§十三-#1

**Files:**
- Modify: `backend/app/services/init_system_templates.py:233-242`（S1 code_template）

该策略用个股数据推断市场情绪阶段。计算逻辑：
1. 计算价格变化率序列的波动率和方向一致性
2. 高波动+持续上涨 → 高潮，低波动+横盘 → 冰点
3. 从波动率变化推断回暖/退潮

```python
'code_template': '''# S1 市场情绪周期策略 — 真实实现
# 基于个股价格行为推断市场情绪阶段
import numpy as np

# 计算收益率
ret = np.diff(close.values) / np.maximum(close.values[:-1], 1e-10)
ret = np.append(ret, 0.0)  # 对齐长度

# 短周期波动率 (5日) vs 长周期波动率 (20日)
vol_short = pd.Series(ret).rolling(5, min_periods=3).std().fillna(0).values
vol_long = pd.Series(ret).rolling(20, min_periods=8).std().fillna(0).values
vol_ratio = np.where(vol_long > 1e-10, vol_short / vol_long, 1.0)

# 方向一致性：最近5日同向运动比例
n = len(close)
recent_ret = ret[-20:] if n >= 20 else ret
direction = np.sign(recent_ret)
consistency = abs(direction.mean())  # 0=震荡, 1=一致

# 动量强度
momentum = (close.iloc[-1] / max(close.iloc[-20], 1e-10) - 1) if n >= 20 else 0.0
momentum = np.clip(momentum, -0.2, 0.2)

# 四阶段判定
vol_ratio_latest = vol_ratio[-1] if len(vol_ratio) > 0 else 1.0
consistency_latest = float(consistency)
vol_long_latest = vol_long[-1] if len(vol_long) > 0 else 0.0
vol_long_median = float(np.median(vol_long[-20:])) if len(vol_long) >= 20 else vol_long_latest
vol_regime = vol_long_latest > vol_long_median * 1.2  # True=高波动

if consistency_latest > 0.6 and momentum > 0.03 and vol_regime:
    phase = "高潮"
    sentiment_score = 80 + int(min(momentum * 100, 15))
    signal = 1
elif consistency_latest > 0.4 and momentum > 0.0 and not vol_regime:
    phase = "回暖"
    sentiment_score = 60 + int(min(momentum * 100, 20))
    signal = 1 if momentum > 0.02 else 0
elif consistency_latest < 0.2 and abs(momentum) < 0.02:
    phase = "冰点"
    sentiment_score = 20 + int(max(vol_ratio_latest * 15, 0))
    signal = 0
else:
    phase = "退潮"
    sentiment_score = max(10, 40 - int(abs(momentum) * 100))
    signal = -1 if momentum < -0.03 else 0
signal = np.clip(signal, -1, 1)'''
```

```python
'code_template': '''# S3 趋势通道识别策略 — 真实实现
# 识别上升/下降/横盘通道，检测通道突破
import numpy as np

n_period = 20
confirm = 3

if len(close) < n_period + confirm:
    signal = 0
else:
    # 计算唐奇安通道
    recent_high = high.iloc[-n_period:].max()
    recent_low = low.iloc[-n_period:].min()
    channel_height = recent_high - recent_low
    mid = (recent_high + recent_low) / 2

    # 当前价格在通道中的位置
    pos = (close.iloc[-1] - recent_low) / max(channel_height, 1e-10)

    # 趋势方向：线性回归斜率
    x = np.arange(n_period)
    y = close.iloc[-n_period:].values
    slope = np.polyfit(x, y, 1)[0] / max(np.mean(y), 1e-10)

    # 突破检测：最近 confirm 根K线连续在通道外
    last_n_close = close.iloc[-confirm:].values
    last_n_high = high.iloc[-confirm:].values
    last_n_low = low.iloc[-confirm:].values

    breakout_up = all(c > recent_high * 0.99 for c in last_n_close[-2:]) and slope > 0.005
    breakout_down = all(c < recent_low * 1.01 for c in last_n_close[-2:]) and slope < -0.005

    if breakout_up:
        signal = 1
    elif breakout_down:
        signal = -1
    elif slope > 0.003:
        signal = 1 if pos > 0.5 else 0
    elif slope < -0.003:
        signal = -1 if pos < 0.5 else 0
    else:
        signal = 0
signal = np.clip(signal, -1, 1)'''
```

```python
'code_template': '''# S4 涨停板短线策略 — 真实实现
# 基于OHLCV检测涨停板质量和连板潜力
import numpy as np

if len(close) < 5:
    signal = 0
else:
    # A股涨停阈值 ~10%（ST 5%，简化统一用 9.5%）
    limit_up_threshold = 0.095
    prev_close = close.shift(1)

    # 检测涨停日
    is_limit_up = (close / prev_close - 1) >= limit_up_threshold

    # 连板计数（最近连续涨停天数）
    consecutive = 0
    for i in range(len(is_limit_up) - 1, -1, -1):
        if is_limit_up.iloc[i]:
            consecutive += 1
        else:
            break

    # 封板质量评估
    if consecutive > 0:
        # 有连板：基于封板强度评分
        latest_up = (close.iloc[-1] / prev_close.iloc[-1] - 1) if not np.isnan(prev_close.iloc[-1]) else 0
        gap_up = (low.iloc[-1] / prev_close.iloc[-1] - 1) if not np.isnan(prev_close.iloc[-1]) else 0

        # 缩量涨停 = 强板，放量涨停 = 有抛压
        avg_vol = volume.iloc[-min(20, len(volume)):-1].mean() if len(volume) > 1 else volume.iloc[-1]
        vol_ratio_today = volume.iloc[-1] / max(avg_vol, 1)

        if consecutive >= 3:
            # 3板以上：高位连板，风险加大
            strength = max(0, 70 - consecutive * 5)
        elif vol_ratio_today < 1.2 and gap_up > 0.01:
            strength = 85  # 缩量+高开 = 强板
        elif vol_ratio_today < 1.5:
            strength = 65  # 温和放量 = 中板
        else:
            strength = 40  # 巨量 = 弱板

        signal = 1 if strength > 50 else 0
    else:
        # 无涨停
        latest_ret = close.iloc[-1] / max(prev_close.iloc[-1], 1e-10) - 1
        if latest_ret > 0.07:
            signal = 0  # 接近涨停但未封板 = 观望
        else:
            signal = 0  # 无涨停信号
signal = np.clip(signal, -1, 1)'''
```

### Task 5: P0-1 — Vibe S6 波浪理论阶段识别策略

**Covers:** 待解决事项§十三-#4

**Files:**
- Modify: `backend/app/services/init_system_templates.py:447-456`（S6 code_template）

通过摆动高低点识别波浪阶段。

```python
'code_template': '''# S6 波浪理论阶段识别策略 — 真实实现
# 识别摆动高低点，判断波浪阶段
import numpy as np

if len(close) < 30:
    signal = 0
else:
    # 寻找摆动极值点（简单的高低点检测，窗口=5）
    window = 5
    closes = close.values
    highs_v = high.values
    lows_v = low.values

    peaks = []  # (index, price)
    troughs = []

    for i in range(window, len(closes) - window):
        if highs_v[i] == max(highs_v[i - window:i + window + 1]):
            peaks.append((i, highs_v[i]))
        if lows_v[i] == min(lows_v[i - window:i + window + 1]):
            troughs.append((i, lows_v[i]))

    if len(peaks) < 2 or len(troughs) < 2:
        signal = 0
    else:
        # 从最近的摆动点判断趋势
        last_peak = peaks[-1][1] if peaks else closes[-1]
        last_trough = troughs[-1][1] if troughs else closes[-1]
        prev_peak = peaks[-2][1] if len(peaks) >= 2 else last_peak
        prev_trough = troughs[-2][1] if len(troughs) >= 2 else last_trough

        # 更高的高点和更高的低点 = 上升趋势
        higher_high = last_peak > prev_peak * 1.02
        higher_low = last_trough > prev_trough * 1.02

        # 更低的低点和更低的高点 = 下降趋势
        lower_high = last_peak < prev_peak * 0.98
        lower_low = last_trough < prev_trough * 0.98

        # 当前价格相对于最近波段的位置
        current = closes[-1]
        recent_range = max(highs_v[-20:]) - min(lows_v[-20:]) if len(closes) >= 20 else 0
        pos_in_wave = (current - min(lows_v[-20:])) / max(recent_range, 1e-10) if recent_range > 0 else 0.5

        if higher_high and higher_low:
            # 上升驱动
            signal = 1 if pos_in_wave < 0.7 else 0
        elif lower_high and lower_low:
            # 下降驱动
            signal = -1 if pos_in_wave > 0.3 else 0
        elif higher_low and lower_high:
            # 调整/收敛
            signal = 0
        else:
            signal = 0
signal = np.clip(signal, -1, 1)'''
```

```python
'code_template': '''# S7 斐波那契时间周期策略 — 真实实现
# 基于波段极值计算斐波那契回调位
import numpy as np

if len(close) < 30:
    signal = 0
else:
    # 寻找最近显著波段（60日窗口内的最高最低）
    lookback = min(60, len(close))
    recent_high = high.iloc[-lookback:].max()
    recent_low = low.iloc[-lookback:].min()
    high_idx = high.iloc[-lookback:].idxmax()
    low_idx = low.iloc[-lookback:].idxmin()

    # 确定波段方向：高出现在低点之后 → 上升波段，反之下降
    if isinstance(high_idx, int):
        high_pos = high_idx
        low_pos = low_idx
    else:
        high_pos = list(close.index).index(high_idx) if high_idx in close.index else 0
        low_pos = list(close.index).index(low_idx) if low_idx in close.index else 0

    up_swing = high_pos > low_pos
    swing_range = abs(recent_high - recent_low)

    if swing_range < recent_low * 0.05:
        # 波动太小
        signal = 0
    else:
        current = close.iloc[-1]
        fib_levels = [0.236, 0.382, 0.5, 0.618, 0.786]

        if up_swing:
            # 上升波段：计算回调位
            fib_prices = [recent_high - lvl * swing_range for lvl in fib_levels]
            # 当前位置在哪个回调位附近
            near_support = False
            for fp in fib_prices[2:]:  # 0.5/0.618/0.786
                if abs(current - fp) / max(swing_range, 1e-10) < 0.05:
                    near_support = True
                    break
            signal = 1 if near_support else 0
        else:
            # 下降波段：计算反弹位
            fib_prices = [recent_low + lvl * swing_range for lvl in fib_levels]
            near_resistance = False
            for fp in fib_prices[2:]:
                if abs(current - fp) / max(swing_range, 1e-10) < 0.05:
                    near_resistance = True
                    break
            signal = -1 if near_resistance else 0
signal = np.clip(signal, -1, 1)'''
```
