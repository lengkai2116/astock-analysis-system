```python
# backend/app/opportunity_atlas/advice_engine.py
"""L6 操作建议层 — 390号方案 §8

根据 final_score + dims_factor + l0 + dim_results 计算动态仓位、
止损位、目标位等操作建议。
"""
from __future__ import annotations
import logging
from typing import Any

logger = logging.getLogger(__name__)


def _safe_float(val: Any, default: float = 0.0) -> float:
    if val is None:
        return default
    try:
        return float(val)
    except (TypeError, ValueError):
        return default


def compute_advice(
    final_score: float,
    dims_factor: dict,
    l0: dict,
    dim_results: dict,
    ts_code: str,
) -> dict:
    """§8.2 动态仓位模型 + §8.3 操作建议输出

    Returns:
        dict: 含 max_position_ratio / hold_only / soft_risks / hard_veto /
              stop_loss_price / target_price / risk_reward_ratio / entry_zone /
              invalidation_conditions / temperature / atr_pct /
              risk_budget_position / final_score / semantic_type /
              consensus_detail / conflict_summary / reliability_summary
    """
    # ── 基础仓位（§8.2 Step 1）──
    if final_score >= 70:
        base = 0.6
    elif final_score >= 55:
        base = 0.4
    elif final_score >= 30:
        base = 0.1
    else:
        base = 0.0

    # ── L0软风险系数连乘（§8.2 Step 2）──
    position = base * l0.get('position_coeff', 1.0)

    # ── 情绪阶段上限（§8.2 Step 3）──
    emotion_cap = l0.get('emotion_position_cap')
    if emotion_cap is not None:
        position = min(position, emotion_cap)

    # ── 风险预算约束（§8.2 Step 4）──
    risk_budget_pos = None
    if dim_results:
        risk_sd = (dim_results.get('risk') or {}).get('status_description') or {}
        _stop_loss = risk_sd.get('support_price')
        _entry = None
        # 尝试从dim_results获取最新价（如果有的话）
        # 注意：真实场景中由调用方传入，此处仅做安全降级
        if _stop_loss and _entry and _entry > _stop_loss:
            _risk_per_share = _entry - _stop_loss
            _max_loss = 1000000.0 * 0.02  # 100万账户，2%规则
            risk_budget_pos = min(0.30, _max_loss / _risk_per_share * _entry / 1000000.0)
            position = min(position, risk_budget_pos)

    # ── 波动率调整（§8.2 Step 5）──
    risk_factor = dims_factor.get('risk', {})
    _atr_pct = _safe_float(risk_factor.get('atr_pct', 0.0) if isinstance(risk_factor, dict) else 0.0)
    # 从dim_results补充
    if not _atr_pct and dim_results:
        _atr_pct = _safe_float(
            (dim_results.get('risk') or {}).get('status_description', {}).get('atr_pct', 0.0)
        )
    if _atr_pct > 0.8:
        position *= 0.6

    # ── 盈亏比调整（§8.2 Step 6）──
    _rr = _safe_float(
        (dim_results.get('risk') or {}).get('status_description', {}).get('rr_value', 1.0)
    ) if dim_results else 1.0
    if _rr > 0:
        position *= min(_rr / 2.0, 1.0)

    # ── 距防守位调整（§8.2 Step 7）──
    _dist_support = _safe_float(
        dims_factor.get('risk', {}).get('dist_to_support_pct', -5.0)
        if isinstance(dims_factor.get('risk'), dict) else -5.0
    )
    if _dist_support > -2:
        position *= 0.6

    # ── 单票上限30%（§8.2 Step 8）──
    position = min(position, 0.30)
    position = max(position, 0.0)

    # ── hold_only覆盖 ──
    hold_only = l0.get('hold_only', False)
    if hold_only:
        position = 0.0

    # ── 组装操作建议（§8.3）──
    advice = {
        'max_position_ratio': round(position, 2),
        'hold_only': hold_only,
        'soft_risks': l0.get('soft_risks', []),
        'hard_veto': l0.get('hard_veto', False),
        'final_score': round(final_score, 1),
        'semantic_type': '',
        'consensus_detail': {},
        'conflict_summary': {},
        'reliability_summary': {},
    }

    # ── 从dim_results提取止损/目标/盈亏比等 ──
    if dim_results:
        risk_sd = (dim_results.get('risk') or {}).get('status_description') or {}
        if risk_sd.get('support_price'):
            advice['stop_loss_price'] = risk_sd['support_price']
        if risk_sd.get('resistance_price'):
            advice['target_price'] = risk_sd['resistance_price']
        if risk_sd.get('rr_value') is not None:
            advice['risk_reward_ratio'] = risk_sd['rr_value']
        if risk_sd.get('invalidation'):
            advice['invalidation_conditions'] = risk_sd['invalidation']
        if risk_sd.get('atr_pct') is not None:
            advice['atr_pct'] = risk_sd['atr_pct']
        emo_sd = (dim_results.get('emotion') or {}).get('status_description') or {}
        if emo_sd.get('temperature') is not None:
            advice['temperature'] = emo_sd['temperature']
        vp_sd = (dim_results.get('volume_price') or {}).get('status_description') or {}
        if vp_sd.get('entry_zone'):
            advice['entry_zone'] = vp_sd['entry_zone']
        if risk_budget_pos is not None:
            advice['risk_budget_position'] = risk_budget_pos

    return advice
```

```python
    def evaluate(self, ts_code: str, dim_results: Optional[dict] = None) -> Optional[dict]:
        """生产环节主流程：原料 → L1 → L0 → L2 → 成品仓行"""
        import time as _time
        _t0 = _time.time()

        tags = self._load_tags(ts_code)
        signals = self._load_signals(ts_code)
        if not tags and not signals:
            return None

        lifecycle = self._signal_lifecycle(ts_code, tags, signals)

        if dim_results:
            dim_engine_results = dim_results
        else:
            dim_engine_results = self._build_dim_engine_results(tags, signals, {}, lifecycle)

        try:
            from app.opportunity_atlas.dimensions.dim8_summary_engine import Dim8SummaryEngine
            dim8 = Dim8SummaryEngine()
            dim_engine_results['summary'] = dim8.evaluate(dims={}, tags=tags, lifecycle={'dim_results': dim_engine_results})
        except Exception as e:
            logger.warning(f"dim8 状态总结失败: {e}")
            dim_engine_results['summary'] = None

        dims = self._convert_to_dims_format(dim_engine_results, tags)

        # BUG-3修复：从dim5引擎提取emotion_phase供_apply_l0使用
        _emotion_phase = ''
        if dim_engine_results:
            _emo_sd = (dim_engine_results.get('emotion') or {}).get('status_description') or {}
            _emotion_phase = str(_emo_sd.get('market_phase', ''))

        l0 = self._apply_l0(ts_code, tags, dims, lifecycle, emotion_phase=_emotion_phase)

        # ═══ 390号新管线分支 ═══
        _engine_version = 'legacy'
        try:
            from app.services.status_config import get_status_engine_config
            _engine_version = get_status_engine_config().get('jud_engine_version', 'legacy')
        except Exception:
            pass

        if _engine_version == 'v390' and dim_engine_results:
            l2 = self._aggregate_v390(tags, dims, l0, lifecycle,
                                      dim_results=dim_engine_results, ts_code=ts_code)
        else:
            l2 = self._aggregate(tags, dims, l0, lifecycle,
                                 dim_results=dim_engine_results, ts_code=ts_code)

        hits = self._detect_registered_signals(tags, signals)
        result = self._assemble(ts_code, dims, lifecycle, l0, l2, hits, dim_engine_results)

        _elapsed = (_time.time() - _t0) * 1000
        if _elapsed > 500:
            logger.info(f"evaluate({ts_code}) 耗时{_elapsed:.0f}ms（>500ms）")

        return result
```

```python
    def _aggregate_v390(self, tags: dict, dims: dict, l0: dict, lifecycle: Optional[dict],
                        dim_results: dict = None, ts_code: str = '') -> dict:
        """390号方案v3.0：L1→L2→L3→L4→L5 完整新管线"""
        dim_results = dim_results or {}

        # L1：维度适配 → 因子格式
        from app.opportunity_atlas.dim_adapter import convert_to_factors
        dims_factor = convert_to_factors(dim_results, tags)

        # L2：可靠性评估
        from app.opportunity_atlas.reliability_assessor import assess
        reliability = assess(dims_factor, dim_results)

        # L3：共识聚合（族级去相关 + 状态依赖权重 + 可靠性加权）
        from app.opportunity_atlas.consensus_engine import compute as consensus_compute
        # 检测市场regime用于选择权重矩阵
        _emotion_phase = 'normal'
        if dim_results:
            _emo_sd = (dim_results.get('emotion') or {}).get('status_description') or {}
            _emo_phase = str(_emo_sd.get('market_phase', ''))
            if _emo_phase in ('ice', 'ebb', 'normal', 'recovery', 'positive', 'climax'):
                _emotion_phase = _emo_phase

        # 使用旧weight_engine获取权重（兼容IC动态权重）
        try:
            from app.opportunity_atlas.weight_engine import get_weights, detect_market_regime, compute_ic_from_history
            regime = detect_market_regime(tags, dims or {})
            ic_values = compute_ic_from_history(self.dm, ts_code) if ts_code else None
            _emotion_quadrant = 'MM'
            if dim_results:
                _emo_q = (dim_results.get('emotion') or {}).get('status_description', {}).get('bociasi_quadrant')
                if _emo_q:
                    _emotion_quadrant = str(_emo_q)
            weights_7dim = get_weights(regime, ic_values, _emotion_quadrant)
        except Exception:
            weights_7dim = {}

        # 将7维度权重映射到6族权重
        _family_weights = self._map_dim_weights_to_family(weights_7dim, _emotion_phase)

        consensus = consensus_compute(dims_factor, reliability, _family_weights,
                                      emotion_phase=_emotion_phase)

        # L4：冲突矩阵
        from app.opportunity_atlas.conflict_matrix import detect as conflict_detect
        conflict = conflict_detect(dims_factor, tags, dim_results,
                                   consensus_rate=abs(consensus.get('consensus_rate', 0)))

        # 硬否决检查（§9.1: L4 fatal_to_veto → wait）
        if l0.get('hard_veto'):
            return {
                'opportunity_state': 'avoid',
                'state_evidence': [l0.get('hard_reason', 'L0硬否决')],
                'consensus_rate': 0.0,
                'display_consensus_rate': 0.0,
                'direction': 'neutral',
                'bullish_dims': 0.0,
                'bearish_dims': 0.0,
                'conflict_evidence': [],
                'fatal_conflicts': [l0.get('hard_reason', '')],
            }

        if conflict.get('fatal_to_veto'):
            return {
                'opportunity_state': 'wait',
                'state_evidence': [f"致命冲突: {', '.join(conflict['fatal_to_veto'])}"],
                'consensus_rate': abs(consensus.get('consensus_rate', 0)),
                'display_consensus_rate': abs(consensus.get('consensus_rate', 0)),
                'direction': consensus.get('direction', 'neutral'),
                'bullish_dims': consensus.get('bull_score', 0),
                'bearish_dims': consensus.get('bear_score', 0),
                'conflict_evidence': conflict.get('warn_for_semantic', []),
                'fatal_conflicts': conflict['fatal_to_veto'],
            }

        # L5：多因子仲裁
        from app.opportunity_atlas.factor_arbiter import arbitrate as factor_arbitrate
        arb_result = factor_arbitrate(consensus, conflict, tags, dims_factor, reliability)

        # L6：操作建议
        from app.opportunity_atlas.advice_engine import compute_advice
        advice = compute_advice(
            final_score=arb_result['final_score'],
            dims_factor=dims_factor,
            l0=l0,
            dim_results=dim_results,
            ts_code=ts_code,
        )
        advice['semantic_type'] = conflict.get('semantic_type', '')
        advice['reliability_summary'] = reliability
        advice['consensus_detail'] = consensus.get('group_details', {})
        advice['conflict_summary'] = {
            'fatal': conflict.get('fatal_to_veto', []),
            'warn': conflict.get('warn_for_semantic', []),
        }

        # 背景周期过滤（保留旧逻辑）
        _bg_evidence = []
        if dim_results and consensus.get('direction') == 'bull':
            structure = dim_results.get('structure') or {}
            sd = structure.get('status_description') or {}
            level_trends = str(sd.get('level_trends', ''))
            if '/周' in level_trends:
                _weekly_part = level_trends.split('/周')[1].split('/')[0].strip()
                if _weekly_part in ('down', '下降', '空'):
                    _cr = abs(consensus.get('consensus_rate', 0))
                    if _cr < 0.80:
                        _bg_evidence.append(f'背景周期过滤：周线{_weekly_part}+日线看多共识{_cr:.0%}<80%→降级')
                        arb_result['opportunity_state'] = 'wait'

        # hold_only覆盖
        if l0.get('hold_only') and arb_result['opportunity_state'] in ('enter', 'light'):
            arb_result['opportunity_state'] = 'wait'
            arb_result['state_evidence'] = ['信号已延伸：只可持有、不新开仓（L0c）'] + arb_result['state_evidence']

        state_evidence = arb_result['state_evidence']
        if _bg_evidence:
            state_evidence = _bg_evidence + state_evidence

        return {
            'opportunity_state': arb_result['opportunity_state'],
            'state_evidence': state_evidence,
            'consensus_rate': abs(consensus.get('consensus_rate', 0)),
            'display_consensus_rate': abs(consensus.get('consensus_rate', 0)),
            'direction': consensus.get('direction', 'neutral'),
            'bullish_dims': consensus.get('bull_score', 0),
            'bearish_dims': consensus.get('bear_score', 0),
            'conflict_evidence': conflict.get('warn_for_semantic', []),
            'fatal_conflicts': conflict.get('fatal_to_veto', []),
            'final_score': arb_result['final_score'],
            'semantic_type': conflict.get('semantic_type', ''),
            'reliability_summary': reliability,
            'consensus_detail': consensus.get('group_details', {}),
            'advice': advice,
        }

    @staticmethod
    def _map_dim_weights_to_family(weights_7dim: dict, emotion_phase: str) -> dict:
        """将7维度权重映射到6族权重（用于consensus_engine）"""
        # 使用consensus_engine内置的STATE_WEIGHTS作为基准
        from app.opportunity_atlas.consensus_engine import STATE_WEIGHTS
        base = STATE_WEIGHTS.get(emotion_phase, STATE_WEIGHTS['normal'])

        # 如果有IC动态权重，按族聚合
        if weights_7dim:
            for family, dims in [
                ('structure_trend', ['structure', 'signal']),
                ('volume_price', ['vp']),
                ('valuation_quality', ['valuation']),
                ('risk', ['risk']),
            ]:
                family_dims = [d for d in dims if d in weights_7dim]
                if family_dims:
                    base[family] = sum(weights_7dim[d] for d in family_dims) / len(family_dims)

            # 主力行为族
            if 'chip_fund' in weights_7dim:
                base['main_behavior'] = weights_7dim['chip_fund']

            # 环境族（从emotion权重推导）
            if 'emotion' in weights_7dim:
                base['environment'] = weights_7dim['emotion']

            # 归一化
            total = sum(base.values())
            if total > 0:
                base = {k: v / total for k, v in base.items()}

        return base
```
